// Title: Find First Palindromic String in the Array
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/find-first-palindromic-string-in-the-array/

class Solution:
    def firstPalindrome(self, words: List[str]) -> str:
        for word in words:
            if word == word[::-1]:
                return word
        return ""
