// Title: Maximum Product of Two Elements in an Array
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/maximum-product-of-two-elements-in-an-array/

class Solution:
    def maxProduct(self, nums: list[int]) -> int:
        nums = sorted(nums)
        return (nums[-1]-1)*(nums[-2]-1)
