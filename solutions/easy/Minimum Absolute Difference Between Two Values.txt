// Title: Minimum Absolute Difference Between Two Values
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/minimum-absolute-difference-between-two-values/

class Solution:
    def minAbsoluteDifference(self, nums: list[int]) -> int:
        for i in range(len(nums)):
            for j in range(i+1,len(nums)):
                if (nums[i] == 1 and nums[j] ==2) or (nums[j] == 1 and nums[i] ==2):

        m = float("inf")
                    m = min(m,abs(i-j), abs(j-i))
        if m == float("inf"):
            return -1
        else:
            return m
