// Title: Richest Customer Wealth
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/richest-customer-wealth/

class Solution:
    def maximumWealth(self, accounts: List[List[int]]) -> int:
        max_miney = 0
        for i in accounts:
            max_miney = max(sum(i), max_miney)
        return max_miney
