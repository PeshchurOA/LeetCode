// Title: Find the Pivot Integer
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/find-the-pivot-integer/

class Solution:
    def pivotInteger(self, n: int) -> int:
        if n == 1:
            return 1
        
        n_list = [x for x in range(1,n+1)]
        for i in range(1,n+1):
            if sum(n_list[:i+1]) == sum(n_list[i:]):
                return i+1
            
        return -1
