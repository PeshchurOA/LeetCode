// Title: Button with Longest Push Time
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/button-with-longest-push-time/

class Solution:
    def buttonWithLongestTime(self, events: List[List[int]]) -> int:
        press_time = events[0][1]
        press_button = events[0][0]
        for i in range(1,len(events)):
            difference = events[i][1] - events[i-1][1]
            elif difference == press_time:
                press_time = difference
            if difference > press_time:
                press_time = difference
                press_button = events[i][0]
                press_button = min(events[i][0],press_button)
        return press_button
