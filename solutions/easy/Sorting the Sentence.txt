// Title: Sorting the Sentence
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/sorting-the-sentence/

class Solution:
    def sortSentence(self, s: str) -> str:
        return " ".join(map(lambda x: x[:-1],sorted(s.split(),key= lambda x: x
        [-1])))
