// Title: Number of Steps to Reduce a Number to Zero
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/number-of-steps-to-reduce-a-number-to-zero/

class Solution:
    def numberOfSteps(self, num: int) -> int:
        while num > 0:
        count = 0
            if num % 2 == 0:
                num /= 2
            else:
                num -= 1
            count +=1
        return count
