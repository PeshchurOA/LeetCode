// Title: Remove Duplicates from Sorted List
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/remove-duplicates-from-sorted-list/

class Solution:
    def deleteDuplicates(self, head: Optional[ListNode]) -> Optional[ListNode]:
        s = set()
        while head:
            s.add(head.val)
            head = head.next
        
        dummy = ListNode(0)
        curr = dummy
        
        for vall in s:
            curr.next = ListNode(vall)
            curr = curr.next
        s = sorted(s)
        return dummy.next
