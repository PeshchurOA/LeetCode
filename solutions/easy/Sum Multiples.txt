// Title: Sum Multiples
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/sum-multiples/

class Solution:
    def sumOfMultiples(self, n: int) -> int:
        result = 0
        for i in range(n+1):
            if i % 3 == 0 or i % 5 == 0 or i % 7 == 0:
                result += i
        return result
