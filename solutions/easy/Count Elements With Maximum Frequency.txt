// Title: Count Elements With Maximum Frequency
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/count-elements-with-maximum-frequency/

class Solution:
    def maxFrequencyElements(self, nums: List[int]) -> int:
        d = {}
        for num in nums:
            d[num] = d.get(num,0)+1
        m = [x[1] for x in sorted(d.items(),key = lambda x: x[1], 
        reverse= True)]
        return m.count(max(m)) * max(m)
