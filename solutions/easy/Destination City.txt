class Solution:
    def destCity(self, paths: list[list[str]]) -> str:
        tmp = {}
        for i,x in paths:
            tmp[i] = tmp.get(i,0)+1
            #tmp[x] = tmp.get(x,0)+1
        for i,x in paths:
            if i not in tmp:
                return i
            if x not in tmp:
                return x
