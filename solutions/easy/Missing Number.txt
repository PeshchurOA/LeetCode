// Title: Missing Number
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/missing-number/

class Solution:
    def missingNumber(self, nums: List[int]) -> int:
        nums[:] = list(sorted(nums))
        for i in range(nums[-1]+1):
            if i not in nums:
                return i
        
        return nums[-1]+1
