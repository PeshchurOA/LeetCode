        result = 0
        for num in range(low,high+1):
            num_= list(map(int,tuple(str(num))))
            l = len(num_)
            
            if l % 2 != 0:
                continue
            q = int(l/2)
            if sum(num_[:q]) == sum(num_[q:]):
                result +=1
        return result
