// Title: Merge Sorted Array
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/merge-sorted-array/

class Solution:
    def merge(self, nums1: List[int], m: int, nums2: List[int], n: int) -> None:
        m = nums1 + nums2


        for i in range(n):
            m.remove(0)
        for i in range(len(nums1)):
            nums1[i] = m[i]
        return nums1
        m.sort()
