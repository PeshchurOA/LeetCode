// Title: Shuffle String
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/shuffle-string/

class Solution:
    def restoreString(self, s: str, indices: List[int]) -> str:
        result = [0 for i in range(len(s))]
        for i in range(len(s)):
            result[indices[i]] = s[i]
        return "".join(result)
